% Common template of the experiment-design schematics (one per experiment).
% Five columns: (1) landscapes and runs, (2) manipulation, (3) programme,
% (4) measured outcomes, (5) analysis.  \expgrid{body height in cm}{t1}..{t5}
% Column left edges: \cA..\cE ; widths \wA..\wE (cm).
% ---------------------------------------------------------------------------
\def\cA{0}\def\wA{3.1}
\def\cB{3.45}\def\wB{5.2}
\def\cC{9.0}\def\wC{2.6}
\def\cD{11.95}\def\wD{2.45}
\def\cE{14.75}\def\wE{2.25}
\tikzset{
  colbody/.style={draw=#1!55, fill=#1!6, rounded corners=3pt, anchor=north west, line width=0.5pt},
  colhead/.style={fill=#1, text=white, font=\footnotesize\sffamily\bfseries, rounded corners=2pt, anchor=north west, inner sep=0pt, minimum height=5.2mm},
  txt/.style={font=\footnotesize\sffamily\sansmath, text=devline, align=flush left, anchor=north west, inner sep=0.5pt, execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  stxt/.style={font=\scriptsize\sffamily\sansmath, text=devline, align=flush left, anchor=north west, inner sep=0.5pt, execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  chev/.style={-{Triangle[length=5pt,width=7pt]}, line width=2.2pt, draw=devline!55},
  chip/.style={draw=#1, fill=#1!15, rounded corners=1.5pt, font=\scriptsize\sffamily\sansmath, text=devline, inner sep=1.4pt, minimum height=4mm, anchor=west, align=center, execute at begin node={\hyphenpenalty=10000\exhyphenpenalty=10000}},
  tst/.style={draw=devline, line width=0.35pt, minimum size=2.6mm, inner sep=0pt, fill=#1},
  tst/.default=white,
}
\colorlet{cA}{devgreen!75!black}
\colorlet{cB}{devamber!90!black}
\colorlet{cC}{devacc!90!black}
\colorlet{cD}{devpurple}
\colorlet{cE}{devline}
\newcommand{\expgrid}[6]{%
  \foreach \x/\w/\t/\c/\n in {\cA/\wA/{#2}/cA/1,\cB/\wB/{#3}/cB/2,\cC/\wC/{#4}/cC/3,\cD/\wD/{#5}/cD/4,\cE/\wE/{#6}/cE/5}{%
    \node[colbody=\c, minimum width=\w cm, minimum height=#1 cm] at (\x,0) {};
    \node[colhead=\c, minimum width=\w cm] at (\x,0) {};
    \node[font=\footnotesize\sffamily\bfseries, text=white, anchor=west, inner sep=0pt] at (\x+0.15,-0.26) {\n\enspace\t};}
  \foreach \a/\b in {3.1/3.45,8.65/9.0,11.6/11.95,14.4/14.75}{\draw[chev] (\a-0.02,-#1/2-0.15) -- (\b+0.06,-#1/2-0.15);}
}
% landscape rows of column 1: \landrow{y}{icon}{name}{runs}
\newcommand{\landrow}[4]{%
  \node[dev, anchor=west] at (\cA+0.1,#1) {#2};
  \node[stxt, anchor=west, text width=1.8cm] at (\cA+1.25,#1) {\textbf{#3}\\#4};}
